	for(int i=0;i<a.len||i<b.len;i++){
	if(a.d[i]<b.d[i]){
		a.d[i+1]--;
		a.d[i]+=10;//TODO
	}	//TODO
	c.d[c.len++]=a.d[i]-b.d[i];
	}
	while(c.len-1>=1&&c.d[c.len-1]==0){

	}

}
高精度乘法：
bign multi(bign a,int b){
	bign c;
	int carry=0;
	for(int i=0;i<a.len;i++){
		int temp=a.d[i]*b+carry;
		c.d[c.len++]=temp%10;
		carry=temp/10;//TODO
	}
	while(carry!=0){
		c.d[c.len++]=carry%10;
		carry/=10;//TODO
	}
	return c;
}

	c.len=a.len;

		r=r*10+a.d[i];
