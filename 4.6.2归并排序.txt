//合并之后排序
void mergeSort(int A[],int left,int right){
	if(left<right){
		int mid=(left+right)/2;//设定一个中点
		mergeSort(A,left,mid);//递归执行对左区间的排序
		mergeSort(A,mid+1,right);//递归执行对右区间的排序
		merge(A,left,mid,mid+1,right);//左区间右区间合并
	}
}
