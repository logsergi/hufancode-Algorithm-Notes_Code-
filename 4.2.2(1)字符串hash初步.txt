字符串映射为整数
int hashFunc(char S[],int len){
int id=0;
for(int i=0;i<len;i++){
id=id*26+S[i]-'A';
}
return id;
}

int hashFunc(char S[],int len){
for(int i=0;i<len;i++){

id=id*52+(S[i]-'A');
}
else if(S[i]>='a'&&S[i]<='z'){
id=id*52+(S[i]-'a')+26
}
