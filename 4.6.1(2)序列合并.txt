int merge(int A[],int B[],int C[],int n,int m){//n，m是数组长度
	int i=0,j=0,index=0;//i,j当作指针用
	while(i<n&&j<m){
		if(A[i]<=B[j]){
			C[index++]=A[i++];//A的元素加入C
		}//TODO
		else{
			C[index++]=B[j++];//B的元素加入C
		}
	}
	while(i<n){
		C[index++]=A[i++];//TODO
	}
	while(j<m){
		C[index++]=B[j++];//TODO
	}//以上是处理剩余元素
	return index;
}
